import { MetadataRoute } from 'next';

/**
 * 검색 엔진 로봇의 접근 권한을 설정하는 화살표 함수
 */
const robots = (): MetadataRoute.Robots => {
  return {
    rules: {
      userAgent: '*', // 모든 검색 로봇(구글, 네이버 등)에 대해
      allow: '/',     // 전체 사이트 접근을 허용
      disallow: [
        '/admin',     // 관리자 페이지가 있다면 검색 노출 제외 (선택사항)
        '/private/',  // 비공개 폴더가 있다면 제외
      ],
    },
    // 구글 로봇이 사이트 지도를 바로 찾을 수 있도록 등록
    sitemap: 'https://beomlog.dev/sitemap.xml',
  };
};

export default robots;
